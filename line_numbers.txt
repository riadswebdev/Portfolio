  32:   const bk = repo.techStack?.backend ?? [];

  44: // ─── 3-D Tilt Card ──────────────────────────────────────────────────────────
  45: function TiltCard({
  46:   children,
  47:   className,
  48: }: {
  49:   children: React.ReactNode;
  50:   className?: string;
  51: }) {
  52:   const ref = useRef<HTMLDivElement>(null);
  53:   const rectRef = useRef<DOMRect | null>(null);
  54: 
  55:   const updateRect = () => {
  56:     if (ref.current) {
  57:       rectRef.current = ref.current.getBoundingClientRect();
  58:     }
  59:   };
  60: 
  61:   const onMove = (e: React.MouseEvent<HTMLDivElement>) => {
  62:     const el = ref.current;
  63:     if (!el) return;
  64:     let rect = rectRef.current;
  65:     if (!rect) {
  66:       rect = el.getBoundingClientRect();
  67:       rectRef.current = rect;
  68:     }
  69:     const x = (e.clientX - rect.left) / rect.width - 0.5;
  70:     const y = (e.clientY - rect.top) / rect.height - 0.5;
  71:     el.style.transition = "transform 0.08s linear";
  72:     el.style.transform = `perspective(900px) rotateY(${x * 24}deg) rotateX(${-y * 24}deg) scale3d(1.03,1.03,1.03)`;
  73:     const glint = el.querySelector<HTMLElement>(".tilt-glint");
  74:     if (glint) {
  75:       glint.style.background = `radial-gradient(circle at ${(x + 0.5) * 100}% ${(y + 0.5) * 100}%, rgba(255,255,255,0.12) 0%, transparent 65%)`;
  76:       glint.style.opacity = "1";
  77:     }
  78:   };
  79: 
  80:   const onLeave = () => {
  81:     const el = ref.current;
  82:     if (!el) return;
  83:     el.style.transition = "transform 0.55s cubic-bezier(.23,1,.32,1)";
  84:     el.style.transform =
  85:       "perspective(900px) rotateY(0deg) rotateX(0deg) scale3d(1,1,1)";
  86:     const glint = el.querySelector<HTMLElement>(".tilt-glint");
  87:     if (glint) glint.style.opacity = "0";
  88:   };
  89: 
  90:   return (
  91:     <div
  92:       ref={ref}
  93:       onMouseMove={onMove}
  94:       onMouseLeave={onLeave}
  95:       className={className}
  96:       style={{ transformStyle: "preserve-3d", willChange: "transform" }}
  97:     >
  98:       {/* glint layer — sits on top, pointer-events:none */}
  99:       <div
 100:         className="tilt-glint absolute inset-0 rounded-3xl pointer-events-none z-10"
 101:         style={{ opacity: 0, transition: "opacity 0.2s" }}
 102:       />
 103:       {children}
 104:     </div>
 105:   );
 106: }
